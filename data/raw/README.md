```bash
python scripts/krisdika_pdf_to_text.py data/sources/krisdika_lpa2541_2568-12-11.pdf data/raw/lpa2541.txt 2026-11-16
python scripts/build_amendments.py data/raw/lpa2541.txt        # → data/amendments.json (ทะเบียนฉบับแก้ไข 9 ฉบับ)
python scripts/verify_amendments.py                            # ทะเบียน vs ราชกิจจาฯ แต่ละฉบับ
python scripts/compare_sources.py data/raw/lpa2541_drthawip.txt data/raw/lpa2541.txt
```
